cmake_minimum_required(VERSION 3.22)
project(neu205_qwen LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT DEFINED LLAMA_SOURCE)
  message(FATAL_ERROR "LLAMA_SOURCE must point to the verified official llama.cpp checkout")
endif()
execute_process(COMMAND git -C "${LLAMA_SOURCE}" rev-parse HEAD OUTPUT_VARIABLE LLAMA_SHA OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT LLAMA_SHA STREQUAL "c6a0ff11a0fe16ac1a2eb569190934593a2b3670")
  message(FATAL_ERROR "Unverified llama.cpp revision")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE}" llama)
add_library(neu205_qwen SHARED qwen_jni.cpp)
target_link_libraries(neu205_qwen PRIVATE llama)
if(ANDROID)
  target_link_options(neu205_qwen PRIVATE "-Wl,-z,max-page-size=16384")
else()
  find_package(JNI REQUIRED)
  target_include_directories(neu205_qwen PRIVATE ${JNI_INCLUDE_DIRS})
endif()
